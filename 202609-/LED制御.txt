#include <FastLED.h>

#define DATA_PIN 6
#define NUM_LEDS 30

CRGB leds[NUM_LEDS];

void setup() {
  FastLED.addLeds<SK6812, DATA_PIN, GRB>(leds, NUM_LEDS);
  FastLED.setBrightness(30);
  delay(2000);
}

void loop() {
  fill_solid(leds, NUM_LEDS, CRGB::Red);
  FastLED.show();
  delay(1000);
}
